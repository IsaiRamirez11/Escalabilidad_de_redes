---- servidor dhcp: configuracion del servicio ----

echo "" > /etc/dhcp/dhcpd.conf
nano /etc/dhcp/dhcpd.conf

archivo:

authoritative;

subnet 20.20.20.0 netmask 255.255.255.0
{
}

subnet 10.10.10.0 netmask 255.255.255.0
{
    range 10.10.10.11 10.10.10.100;
    option routers 10.10.10.254;
   # option domain-name-servers 20.20.20.254;
    option domain-name "isai.mx";
    option broadcast-address 10.10.10.255;
    default-lease-time 600;
    max-lease-time 7200;
}

levantar servicio:

service isc-dhcp-server start
service isc-dhcp-server status

! solo si status dice que NO esta corriendo:
! abre este archivo y deja la linea asi: INTERFACESv4="eth0"
nano /etc/default/isc-dhcp-server
service isc-dhcp-server restart
